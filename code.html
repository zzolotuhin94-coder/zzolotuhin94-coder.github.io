// The allowed transition graph for reservations.status. Before this file
// existed, ANY valid status could move to ANY other one ('completed' back
// to 'pending', for example). Self-transitions stay legal for 'confirmed'
// and 'arrived' — that is the re-seating path (same status, new tables).
const TRANSITIONS = {
  pending:   ['confirmed', 'rejected', 'cancelled'],
  confirmed: ['confirmed', 'arrived', 'no_show', 'cancelled'],
  arrived:   ['arrived', 'completed', 'cancelled'],
  completed: [],
  rejected:  [],
  cancelled: [],
  no_show:   [],
};

const ALL_STATUSES = Object.keys(TRANSITIONS);
const TERMINAL_STATUSES = ALL_STATUSES.filter((s) => TRANSITIONS[s].length === 0);

function isValidTransition(fromStatus, toStatus) {
  if (!TRANSITIONS[fromStatus]) return false;
  return TRANSITIONS[fromStatus].includes(toStatus);
}

module.exports = { TRANSITIONS, ALL_STATUSES, TERMINAL_STATUSES, isValidTransition };

// server/routes/reservations.js — PATCH /api/reservations/:id (excerpt)
async function handleReservationStatusUpdate(req, res, forcedStatus) {
  const status = forcedStatus || req.body.status;
  if (!ALL_STATUSES.includes(status)) {
    return res.status(400).json({ error: `Unknown status: ${status}` });
  }

  // The ownership scope (partner_id) is part of THIS query, not only of the
  // final UPDATE — a partner can't even probe the status of someone else's
  // booking through an error message.
  const { rows } = await pool.query(
    `SELECT status FROM reservations WHERE id = $1
       AND restaurant_id IN (SELECT restaurant_slug FROM venues WHERE partner_id = $2)`,
    [req.params.id, req.partner.partnerId]
  );
  if (rows.length === 0) return res.status(404).json({ error: 'Not found.' });

  const current = rows[0].status;
  if (!isValidTransition(current, status)) {
    // 409, not 400: the request is well-formed, the resource's state forbids it.
    return res.status(409).json({ error: `Cannot go from ${current} to ${status}.` });
  }
  // ...table checks, one transaction, then broadcast after COMMIT
}

Why it’s built this way

Reservation state machine

  • The policy is data, not a pile of if-statements. One object is the whole lifecycle — easy to review, easy to test, impossible to half-apply.
  • Terminal states are derived, not hand-listed: add a state and the list stays correct.
  • Illegal moves return 409 Conflict — the request is valid, the booking’s state forbids it. The app can tell “you can’t do that now” from “you sent garbage”.
  • Ownership lives in the same query, so a partner can’t even learn the status of another venue’s booking from an error message.

// Every booking occupies a fixed window (default 2 h). One helper decides
// whether two windows collide — used by availability, the floor-plan picker
// and the final confirm, so all three always agree.
const DEFAULT_DURATION_MINUTES = 120;

function timeToMinutes(time) {
  const [h, m] = time.split(':').map(Number);
  return h * 60 + m;
}

// Half-open intervals: 18:00–20:00 and 20:00–22:00 do NOT overlap,
// so back-to-back seatings on the same table stay possible.
function windowsOverlap(startA, durA, startB, durB) {
  const aStart = timeToMinutes(startA);
  const aEnd = aStart + durA;
  const bStart = timeToMinutes(startB);
  const bEnd = bStart + durB;
  return aStart < bEnd && bStart < aEnd;
}

// server/routes/reservations.js
// A table can be linked directly (reservations.table_id) or through the
// join table (merged tables for big groups) — both are checked.
async function isTableAvailable(tableId, date, startTime, endTime, excludeId) {
  const result = await pool.query(
    `SELECT DISTINCT r.id, r.reservation_time, r.duration_minutes
     FROM reservations r
     LEFT JOIN reservation_tables rt ON rt.reservation_id = r.id
     WHERE r.reservation_date = $1
       AND r.status IN ('confirmed', 'arrived')
       AND (r.table_id = $2 OR rt.table_id = $2)
       AND ($3::uuid IS NULL OR r.id != $3)`,
    [date, tableId, excludeId || null]
  );

  const requested = Math.max(1, timeToMinutes(endTime) - timeToMinutes(startTime));
  return !result.rows.some((row) =>
    windowsOverlap(startTime, requested, row.reservation_time,
      row.duration_minutes || DEFAULT_DURATION_MINUTES)
  );
}

// "Unified lock": auto-picked AND hand-picked tables go through the same
// check at the moment of confirming — not only when the picker opened.
// That closes the gap where two near-simultaneous confirms took one table.
for (const tableId of tableIds) {
  const free = await isTableAvailable(tableId, target.reservation_date,
    target.reservation_time, endTime, req.params.id);
  if (!free) {
    return res.status(409).json({ error: 'Table already taken for this time.' });
  }
}

Why it’s built this way

No double bookings

  • Half-open time windows — 18:00–20:00 and 20:00–22:00 don’t collide, so a table can be turned twice a night.
  • One overlap helper for every path: the guest’s availability view, the host’s table picker and the final confirm can never disagree.
  • Checked again at the moment of confirming, not only when the picker opened — two hosts tapping at once can’t hand out the same table.
  • Merged tables for large groups are covered through the join table, not special-cased.

// Postgres UPDATE → LISTEN/NOTIFY → this module → a Socket.io room.
// Rooms are scoped as narrowly as the data requires:
//   destination:<id>  every business in one resort (floor plans, queues)
//   venue:<id>        exactly one house (room-service orders)
//   user:<id>         exactly one guest (their own reservation updates)
const { Server } = require('socket.io');

let io = null;

function initSocket(httpServer) {
  io = new Server(httpServer, { /* CORS options omitted in this excerpt */ });

  io.on('connection', (socket) => {
    socket.on('subscribe', (destinationId) => {
      if (typeof destinationId === 'string') socket.join(`destination:${destinationId}`);
    });
    socket.on('subscribe_user', (userId) => {
      if (typeof userId === 'string') socket.join(`user:${userId}`);
    });
    // An order names a room number and an amount, so it must only reach the
    // house it belongs to — never the resort-wide room.
    socket.on('subscribe_venue', (venueId) => {
      if (typeof venueId === 'string') socket.join(`venue:${venueId}`);
    });
    socket.on('unsubscribe_venue', (venueId) => {
      if (typeof venueId === 'string') socket.leave(`venue:${venueId}`);
    });
  });

  return io;
}

// Events carry identifiers and a status, never the full record: clients
// treat them as "refetch now" and load details over the authenticated
// REST route. The database stays the single source of truth.
function broadcastTableStatusChange(payload, destinationId) {
  if (!io) return;
  const target = destinationId ? io.to(`destination:${destinationId}`) : io;
  target.emit('table_status_changed', payload); // { venueId, tableIds, status }
}

function broadcastReservationToUser(userId, payload) {
  if (!io || !userId) return;
  io.to(`user:${userId}`).emit('reservation_changed', payload);
}

// No room, no broadcast: better no live update (the app reloads on open)
// than a room number sent to everyone connected.
function broadcastRoomServiceOrder(venueId, payload) {
  if (!io || !venueId) return;
  io.to(`venue:${venueId}`).emit('room_service_order', payload);
}

module.exports = {
  initSocket,
  broadcastTableStatusChange,
  broadcastReservationToUser,
  broadcastRoomServiceOrder,
};

Why it’s built this way

Socket.io rooms

  • Rooms are as narrow as the data: resort-wide for floor plans, one venue for room-service orders, one guest for their own bookings.
  • Events are signals, not data. They carry IDs and a status; clients refetch through authenticated REST, so the database stays the single source of truth.
  • No room, no broadcast — a missing ID means no live update rather than a room number sent to everyone.
  • The result: a booking lands on the restaurant’s tablet in well under a second, without polling.

// No login (any guest can ask), but every call hits a paid API — so a
// per-IP rate limiter on the route plus hard caps on what one call may cost.
const CONCIERGE_MAX_MESSAGES = 40;
const CONCIERGE_MAX_MESSAGE_LENGTH = 4000;

app.post('/api/concierge/chat', conciergeChatLimiter, async (req, res) => {
  const { messages, hotelName, guestName, locale, reservationId } = req.body;

  // Optional, but if sent it must belong to the caller (no writing into others' threads).
  let userId = null;
  if (req.body?.userId != null) {
    userId = await requireGuestId(req, res, req.body.userId);
    if (!userId) return;
  }

  if (!Array.isArray(messages) || messages.length === 0) {
    return res.status(400).json({ error: 'messages array is required' });
  }
  if (messages.length > CONCIERGE_MAX_MESSAGES ||
      messages.some((m) => typeof m?.content !== 'string' ||
                           m.content.length > CONCIERGE_MAX_MESSAGE_LENGTH)) {
    return res.status(400).json({ error: 'Conversation too long.' });
  }

  const apiKey = process.env.ANTHROPIC_API_KEY; // server-side only, never in the app
  if (!apiKey) return res.status(503).json({ error: 'Concierge is not configured.' });

  try {
    const response = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': apiKey,
        'anthropic-version': '2023-06-01',
      },
      body: JSON.stringify({
        model: 'claude-haiku-4-5-20251001',
        max_tokens: 400,
        // Live data: verified venues + today's open/closed status (cached 5 min).
        system: await buildConciergeSystemPrompt({ hotelName, guestName, locale }),
        messages: messages.map((m) => ({ role: m.role, content: m.content })),
      }),
    });

    if (!response.ok) {
      // The log gets the real cause (bad key, outage, rate limit); the guest a neutral line.
      const described = describeAnthropicFailure(response.status, await response.text());
      console.error(`[concierge] ${response.status} ${described.reason}`);
      // Pass 429 through so the app can say "try again" instead of "broken".
      const status = response.status === 429 ? 429 : 502;
      return res.status(status).json({ error: described.guestMessage });
    }

    const data = await response.json();
    const reply = data.content?.find((b) => b.type === 'text')?.text ?? '';
    const lastUser = [...messages].reverse().find((m) => m.role === 'user');
    await persistConciergeExchange({
      userId, reservationId, userText: lastUser?.content, assistantText: reply,
    });
    res.json({ reply });
  } catch (err) {
    res.status(502).json({ error: 'The concierge is unavailable right now.' });
  }
});

Why it’s built this way

Claude AI concierge

  • The API key never leaves the server — the mobile app only talks to this route.
  • Cost is bounded twice: a per-IP rate limiter on the route, plus caps on message count and length for every call.
  • Failures are translated, not swallowed. The log gets the real cause (bad key, outage, rate limit); the guest gets a friendly sentence; 429 passes through so the app can say “try again”.
  • The prompt is built from live data — only verified, bookable venues and today’s open/closed status, cached for five minutes.

// One canvas store for tables, merged groups and zones. Deliberately React
// Context here: this state is local to the floor-plan dashboard, so it does
// not justify a global store (businessStore.ts is the Zustand one).

type TableStatusEvent = { venueId: string; tableIds: string[]; status: TableStatus };

// Plan coordinates are fractions 0..1 — the server accepts nothing else.
const toPlanFraction = (v: number) => Math.min(0.98, Math.max(0.02, v));

export function FloorPlanStoreProvider({ children }: { children: React.ReactNode }) {
  const [tables, setTables] = useState<Table[]>([]);
  const [venueId, setVenueId] = useState<string | null>(null);

  // Real-time sync: every reservation status change with tables attached
  // emits table_status_changed. The socket room is resort-wide, so the
  // payload is filtered down to THIS venue before anything re-renders.
  useEffect(() => {
    if (!venueId) return undefined;
    return onTableStatusChanged((payload: TableStatusEvent) => {
      if (payload.venueId !== venueId || !payload.tableIds?.length) return;
      setTables((prev) =>
        prev.map((tb) =>
          payload.tableIds.includes(tb.id) ? { ...tb, status: payload.status } : tb)
      );
    });
  }, [venueId]);

  // Optimistic move: the table jumps immediately, the PATCH runs fire-and-
  // forget. Called once from the pan gesture's onEnd — never per frame —
  // so dragging can't flood the API. A failed save is logged, not rolled
  // back: yanking a table back mid-service is worse than a lost position.
  const moveTable = useCallback((id: string, globalX: number, globalY: number) => {
    const x = toPlanFraction(globalX);
    const y = toPlanFraction(globalY);
    setTables((prev) =>
      prev.map((tb) => (tb.id === id ? { ...tb, globalX: x, globalY: y } : tb)));
    updateTablePosition(id, x, y).catch((err) => {
      console.warn('[floorPlanStore] position not saved:', (err as Error).message);
    });
  }, []);

  // Merged tables move as one: same fractional delta for every member,
  // so their arrangement relative to each other is preserved exactly.
  const moveGroupBy = useCallback((groupId: string, dx: number, dy: number) => {
    setTables((prev) => prev.map((tb) =>
      tb.groupId === groupId
        ? { ...tb,
            globalX: toPlanFraction(tb.globalX + dx),
            globalY: toPlanFraction(tb.globalY + dy) }
        : tb
    ));
  }, []);

  const value = useMemo(
    () => ({ tables, venueId, moveTable, moveGroupBy /* …more actions */ }),
    [tables, venueId, moveTable, moveGroupBy]
  );
  return (
    <FloorPlanStoreContext.Provider value={value}>{children}</FloorPlanStoreContext.Provider>
  );
}

Why it’s built this way

Floor-plan store

  • Optimistic by design: a dragged table moves instantly; the save runs once when the gesture ends, never per frame.
  • No rollback mid-service. A failed save is logged; snapping a table back while guests are being seated would be worse.
  • Coordinates are fractions (0..1), clamped before they’re stored — the same plan renders correctly on any tablet.
  • Context, not a global store: this state belongs to one screen. Live updates are filtered to this venue before anything re-renders.

// Zustand instead of React Context: the business switcher in the header AND
// the fetch layer (utils/api.js, outside the component tree) both need the same
// activeBusinessId — a Context provider could never reach it.
import { create } from 'zustand';
import AsyncStorage from '@react-native-async-storage/async-storage';

export type BusinessCategory = 'restaurant' | 'hotel' | 'taxi' | 'activity';
export type PartnerRole = 'owner' | 'manager' | 'hostess' | 'driver';

export type Business = {
  id: string;
  companyName: string;
  destinationId: string;
  role?: PartnerRole;             // must match PARTNER_ROLES on the server
  categories?: BusinessCategory[]; // a hotel with a restaurant has both
};

const ACTIVE_BUSINESS_KEY = '@alpengo_business/active_business_id';

interface BusinessState {
  businesses: Business[];
  activeBusinessId: string | null;
  hydrated: boolean;
  setBusinesses: (businesses: Business[], activeId: string | null) => void;
  setActiveBusinessId: (id: string | null) => Promise<void>;
  hydrate: () => Promise<void>;
  clear: () => void;
}

export const useBusinessStore = create<BusinessState>((set, get) => ({
  businesses: [],
  activeBusinessId: null,
  hydrated: false,

  setBusinesses: (businesses, activeId) => {
    set({ businesses, activeBusinessId: activeId });
    if (activeId) AsyncStorage.setItem(ACTIVE_BUSINESS_KEY, activeId);
  },

  // Persists the choice LOCALLY only (which tab was open last). Real
  // authorization happens on the server via /switch-business, which
  // re-scopes the token — this function never touches the token.
  setActiveBusinessId: async (id) => {
    set({ activeBusinessId: id });
    if (id) await AsyncStorage.setItem(ACTIVE_BUSINESS_KEY, id);
    else await AsyncStorage.removeItem(ACTIVE_BUSINESS_KEY);
  },

  hydrate: async () => {
    if (get().hydrated) return;
    const lastActiveId = await AsyncStorage.getItem(ACTIVE_BUSINESS_KEY);
    set({ activeBusinessId: lastActiveId, hydrated: true });
  },

  clear: () => {
    set({ businesses: [], activeBusinessId: null });
    AsyncStorage.removeItem(ACTIVE_BUSINESS_KEY);
  },
}));

Why it’s built this way

Zustand business store

  • Zustand where it earns its place: the fetch layer lives outside React and still needs the active business — a Context provider can’t reach it.
  • The client never decides access. It only remembers which business was open; the server re-scopes the token on every switch.
  • Types mirror the server: categories match the database constraint, roles match the API’s role guards.
  • Small, explicit actions — no reducers or boilerplate, and easy to reset on logout.
